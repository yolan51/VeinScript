Bundle EntitiesBringRows  author=you
|--- Need Vein.Console
|--- Shape $Row
|    |--- Field title: string
|    `--- Field rank: int
|--- Mark #Row
|--- Builder Row
|    |--- Include $Row
|    `--- Mark #Row
|--- Shard Seed
|    `--- RunOnce
|         |--- Let ranks = List
|         |    |--- Int 4
|         |    |--- Int 1
|         |    |--- Unary -
|         |    |    `--- Int 2
|         |    |--- Int 3
|         |    `--- Int 1
|         `--- Ordered by &Row.rank
|              |--- Bring Row
|              |    |--- Arg = Str "literal"
|              |    `--- Arg = Int 0
|              `--- Target ranks as r
|                   `--- If
|                        |--- Cond = Binary >
|                        |    |--- Ref r
|                        |    `--- Unary -
|                        |         `--- Int 2
|                        `--- Then
|                             `--- Bring Row
|                                  |--- Arg = Concat
|                                  |    |--- Str "src"
|                                  |    |--- Ref Index
|                                  |    `--- Str ""
|                                  `--- Arg = Ref r
`--- Shard Show
     `--- Settled
          |--- Call print
          |    `--- Str "-- spawn order, which is now rank order --"
          |--- Target $Row #Row as r
          |    `--- Call print
          |         `--- Concat
          |              |--- Str "   e"
          |              |--- Ref Entity
          |              |--- Str "  rank "
          |              |--- Path r.Row.rank
          |              |--- Str "  "
          |              |--- Path r.Row.title
          |              `--- Str ""
          |--- Call print
          |    `--- Str "-- ties keep the order the rows arrived in --"
          |--- Target $Row #Row as r
          |    `--- If
          |         |--- Cond = Binary ==
          |         |    |--- Path r.Row.rank
          |         |    `--- Int 1
          |         `--- Then
          |              `--- Call print
          |                   `--- Concat
          |                        |--- Str "   "
          |                        |--- Path r.Row.title
          |                        `--- Str ""
          |--- Call print
          |    `--- Str "-- the filtered row never arrived --"
          `--- Target $Row #Row as r
               `--- If
                    |--- Cond = Binary <
                    |    |--- Path r.Row.rank
                    |    `--- Int 0
                    `--- Then
                         `--- Call print
                              `--- Concat
                                   |--- Str "   should not appear: "
                                   |--- Path r.Row.title
                                   `--- Str ""
