|         |    |--- Arg = Str "Footer"

|         |    |--- Arg = Ref base

|         |    |--- Arg = Int 1
|         |    `--- Arg = Int 9

|              |--- Arg = Str "salu"
|              |--- Arg = Ref base
|              |--- Arg = Ref base
|              |--- Arg = Ref base
|              |--- Arg = Ref base
|              |--- Arg = Ref base
|              `--- Arg = Ref base
